namespace App;

use DateTime;
use DateTimeZone;

class Rad
{
    private string $role = 'developer';

    private array $currentlyBuilding = [
        'web projects',
        'Roblox projects',
        'random experiments'
    ];

    private array $currentlyLearning = [
        'Linux internals',
        'system optimization',
        'AI-assisted development',
        'game development'
    ];

    private array $stack = [
        'TypeScript',
        'JavaScript',
        'React',
        'Next.js',
        'PHP',
        'Symfony',
        'MySQL',
        'PostgreSQL'
    ];

    private array $aboutMe = [
        'languages' => ['french', 'english'],
        'os' => 'CachyOS / Arch Linux',
        'desktop' => 'KDE Plasma',
        'hobbies' => ['coding', 'gaming', 'linux', 'music'],
        'games' => ['Roblox'],
    ];

    private function isSleeping(): bool
    {
        $now = new DateTime('now', new DateTimeZone('Europe/Paris'));
        $hour = (int) $now->format('H');

        return $hour >= 1 && $hour < 10;
    }

    public function getBio(): array
    {
        return [
            'role' => $this->role,
            'building' => $this->isSleeping()
                ? ['sleeping']
                : $this->currentlyBuilding,
            'learning' => $this->currentlyLearning,
            'stack' => $this->stack,
            'aboutme' => $this->aboutMe,
        ];
    }
}
